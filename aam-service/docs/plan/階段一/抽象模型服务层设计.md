## 📋 设计概述

### 设计目标

实现一个**抽象模型服务层**，支持多种模型服务后端（Ollama、vLLM、OpenAI API 等），通过配置切换，无需修改代码。

### 核心设计理念

**Provider Pattern（提供者模式）**：
- 抽象模型服务接口（`IModelProvider`）
- 多种后端实现（Ollama、vLLM、OpenAI 等）
- 统一模型服务（`UnifiedModelService`）
- Provider 工厂（`ModelProviderFactory`）

## 📝 接口定义

### IModelProvider 接口

```python
# src/core/interfaces/i_model_provider.py

from abc import ABC, abstractmethod
from typing import Optional, Dict, Any
from enum import Enum

class ModelProviderType(str, Enum):
    """模型服务提供商类型"""
    OLLAMA = "ollama"
    VLLM = "vllm"
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    CUSTOM = "custom"  # 自定义 API


class IModelProvider(ABC):
    """模型服务提供商抽象接口"""
    
    @abstractmethod
    async def generate(
        self,
        prompt: str,
        system_prompt: Optional[str] = None,
        temperature: float = 0.7,
        max_tokens: int = 2048,
        **kwargs
    ) -> str:
        """
        生成文本
        
        Args:
            prompt: 用户提示
            system_prompt: 系统提示（可选）
            temperature: 温度参数
            max_tokens: 最大 token 数
            **kwargs: 其他参数
            
        Returns:
            生成的文本
        """
        pass
    
    @abstractmethod
    async def check_available(self) -> bool:
        """
        检查服务是否可用
        
        Returns:
            True 如果可用，False 否则
        """
        pass
    
    @property
    @abstractmethod
    def provider_type(self) -> ModelProviderType:
        """返回提供商类型"""
        pass
```

## 🔧 Provider 实现

### Ollama Provider

```python
# src/infrastructure/ai/providers/ollama_provider.py

from langchain_community.llms import Ollama
from src.core.interfaces.i_model_provider import IModelProvider, ModelProviderType

class OllamaProvider(IModelProvider):
    """Ollama 模型服务提供商"""
    
    def __init__(
        self,
        model_name: str,
        base_url: str = "http://localhost:11434",
        timeout: int = 120,
    ):
        self.model_name = model_name
        self.base_url = base_url
        self.timeout = timeout
        self.llm = Ollama(
            model=model_name,
            base_url=base_url,
            timeout=timeout,
        )
    
    @property
    def provider_type(self) -> ModelProviderType:
        return ModelProviderType.OLLAMA
    
    async def generate(self, prompt: str, **kwargs) -> str:
        return await self.llm.ainvoke(prompt)
    
    async def check_available(self) -> bool:
        # 检查 Ollama 服务是否可用
        import httpx
        try:
            async with httpx.AsyncClient(timeout=5.0) as client:
                response = await client.get(f"{self.base_url}/api/tags")
                return response.status_code == 200
        except:
            return False
```

### vLLM Provider

```python
# src/infrastructure/ai/providers/vllm_provider.py

import httpx
from src.core.interfaces.i_model_provider import IModelProvider, ModelProviderType

class VLLMProvider(IModelProvider):
    """vLLM 模型服务提供商（OpenAI 兼容 API）"""
    
    def __init__(
        self,
        model_name: str,
        api_base_url: str = "http://localhost:8000/v1",
        api_key: Optional[str] = None,
        timeout: int = 120,
    ):
        self.model_name = model_name
        self.api_base_url = api_base_url
        self.api_key = api_key
        self.timeout = timeout
    
    @property
    def provider_type(self) -> ModelProviderType:
        return ModelProviderType.VLLM
    
    async def generate(self, prompt: str, **kwargs) -> str:
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            headers = {}
            if self.api_key:
                headers["Authorization"] = f"Bearer {self.api_key}"
            
            messages = []
            if kwargs.get("system_prompt"):
                messages.append({"role": "system", "content": kwargs["system_prompt"]})
            messages.append({"role": "user", "content": prompt})
            
            response = await client.post(
                f"{self.api_base_url}/chat/completions",
                headers=headers,
                json={
                    "model": self.model_name,
                    "messages": messages,
                    "temperature": kwargs.get("temperature", 0.7),
                    "max_tokens": kwargs.get("max_tokens", 2048),
                }
            )
            response.raise_for_status()
            return response.json()["choices"][0]["message"]["content"]
    
    async def check_available(self) -> bool:
        # 检查 vLLM 服务是否可用
        import httpx
        try:
            async with httpx.AsyncClient(timeout=5.0) as client:
                response = await client.get(f"{self.api_base_url}/models")
                return response.status_code == 200
        except:
            return False
```

## 🏭 Provider 工厂

```python
# src/infrastructure/ai/providers/provider_factory.py

from src.core.interfaces.i_model_provider import IModelProvider, ModelProviderType
from src.config.settings import ModelServiceSettings
from src.infrastructure.ai.providers.ollama_provider import OllamaProvider
from src.infrastructure.ai.providers.vllm_provider import VLLMProvider

class ModelProviderFactory:
    """模型服务提供商工厂"""
    
    @staticmethod
    def create_provider(settings: ModelServiceSettings) -> IModelProvider:
        """
        根据配置创建模型服务提供商
        
        Args:
            settings: 模型服务配置
            
        Returns:
            模型服务提供商实例
        """
        if settings.provider_type == ModelProviderType.OLLAMA:
            return OllamaProvider(
                model_name=settings.ollama_model_name or settings.model_name,
                base_url=settings.api_base_url,
                timeout=settings.timeout,
            )
        elif settings.provider_type == ModelProviderType.VLLM:
            return VLLMProvider(
                model_name=settings.model_name,
                api_base_url=settings.vllm_api_base_url or settings.api_base_url,
                api_key=settings.api_key,
                timeout=settings.timeout,
            )
        # ... 其他 Provider
        else:
            raise ValueError(f"不支持的模型服务提供商: {settings.provider_type}")
```

## ⚙️ 配置设计

### ModelServiceSettings

```python
# src/config/settings.py

from enum import Enum

class ModelProviderType(str, Enum):
    OLLAMA = "ollama"
    VLLM = "vllm"
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    CUSTOM = "custom"

class ModelServiceSettings(BaseSettings):
    """模型服务配置"""
    
    # 服务提供商类型
    provider_type: ModelProviderType = Field(
        default=ModelProviderType.OLLAMA,
        alias="MODEL_PROVIDER_TYPE"
    )
    
    # 模型名称
    model_name: str = Field(
        default="deepseek-r1:8b",
        alias="MODEL_NAME"
    )
    
    # API 配置（通用）
    api_base_url: str = Field(
        default="http://localhost:11434",
        alias="MODEL_API_BASE_URL"
    )
    api_key: Optional[str] = Field(
        default=None,
        alias="MODEL_API_KEY"
    )
    timeout: int = Field(default=120, alias="MODEL_TIMEOUT")
    
    # Ollama 特定配置
    ollama_model_name: str = Field(
        default="deepseek-r1:8b",
        alias="OLLAMA_MODEL_NAME"
    )
    
    # vLLM 特定配置
    vllm_api_base_url: str = Field(
        default="http://localhost:8000/v1",
        alias="VLLM_API_BASE_URL"
    )
    
    # OpenAI 兼容 API 配置
    openai_api_base_url: Optional[str] = Field(
        default=None,
        alias="OPENAI_API_BASE_URL"
    )
    openai_api_key: Optional[str] = Field(
        default=None,
        alias="OPENAI_API_KEY"
    )
```

## 🔄 使用流程

### 1. 配置 Provider

```bash
# .env
MODEL_PROVIDER_TYPE=ollama
MODEL_NAME=eb-mm
MODEL_API_BASE_URL=http://localhost:11434
```

### 2. 创建 Provider

```python
from src.config.settings import get_settings
from src.infrastructure.ai.providers.provider_factory import ModelProviderFactory

settings = get_settings()
provider = ModelProviderFactory.create_provider(settings.model_service)
```

### 3. 创建统一模型服务

```python
from src.infrastructure.ai.unified_model_service import UnifiedModelService

unified_service = UnifiedModelService(
    provider=provider,
    model_name=settings.model_service.model_name,
)
```

### 4. 使用统一模型服务

```python
# 在业务逻辑中使用
knowledge = await unified_service.extract_knowledge(
    text="Apple is a technology company.",
    user_id="user123",
    session_id="session456",
)
```

## ✅ 优势

1. **灵活性**: 通过配置切换 Provider，无需修改代码
2. **可扩展性**: 新增 Provider 只需实现接口
3. **解耦**: 业务逻辑与具体 Provider 解耦
4. **测试友好**: 可 Mock Provider 进行测试
5. **未来兼容**: 支持独立模型服务、云服务等
